117 async def get_usage_buckets_page( 

118 self,

127 ) -> UsageBucketPage: 

128 context_filters = self._context_filters(user_id=user_id, endpoint=endpoint, key_id=key_id)

154 """Filters valid on both the observations and scores-numeric views, so both queries aggregate the same requests. 

155 

156 The scores views cannot filter on metadata, so the key is matched through the trace tag set in start_record. 

157 """ 

158 filters: list[dict] = [{"column": "userId", "operator": "=", "value": str(user_id), "type": "string"}]

161 if key_id is not None: 

162 filters.append({"column": "tags", "operator": "any of", "value": [cls._key_tag(key_id)], "type": "arrayOptions"})

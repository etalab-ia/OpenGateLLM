11_KWH_SCORE_COLUMN = "kWh" 

12_KGCO2EQ_SCORE_COLUMN = "kgCO2eq"

15class LangfuseRequestTracer(RequestTracer): 

16 """Traces every model-forward request, failures included. The usage table holds consumption only, so it is the one 

17 place where a failed request leaves a trace.""" 

18 

19 def __init__(self, client: Langfuse) -> None: 

20 self.client = client 

21 self._observation = None 

22 

23 def start(self, record: UsageRecord) -> None: 

24 # the SDK takes the start time, the user and the tags only at creation, so the observation starts with the request 

25 try: 

26 with propagate_attributes(user_id=str(record.user_id), tags=[f"key_id:{record.key_id}"]): 

27 self._observation = self.client.start_observation( 

28 trace_context={"trace_id": record.request_id}, 

29 as_type="generation", 

30 name=record.endpoint, 

31 ) 

32 except Exception: 

33 logger.exception("Failed to start the Langfuse trace.") 

34 

35 def end(self, record: UsageRecord) -> None: 

36 if self._observation is None: 

37 return 

38 

39 try: 

40 self._observation.update(**self._build_update(record)) 

41 if (usage := record.usage) is not None: 

42 self._observation.score(name=_KWH_SCORE_COLUMN, value=usage.impacts.kWh, data_type="NUMERIC") 

43 self._observation.score(name=_KGCO2EQ_SCORE_COLUMN, value=usage.impacts.kgCO2eq, data_type="NUMERIC") 

44 except Exception: 

45 logger.exception("Failed to update the Langfuse trace.") 

46 

47 try: 

48 self._observation.end() 

49 except Exception: 

50 logger.exception("Failed to end the Langfuse trace.") 

51 

52 def _build_update(self, record: UsageRecord) -> dict: 

53 update: dict = { 

54 "model": record.router_name, 

55 "metadata": { 

56 "router_id": record.router_id, 

57 "router_name": record.router_name, 

58 "user_email": record.user_email, 

59 "key_id": str(record.key_id), 

60 "key_name": record.key_name, 

61 "provider_id": record.provider_id, 

62 "provider_model_name": record.provider_model_name, 

63 "status": record.status, 

64 "latency": record.latency, 

65 }, 

66 } 

67 

68 if record.status is None or record.status // 100 != 2 or record.error is not None: 

69 update["level"] = "ERROR" 

70 update["status_message"] = record.error 

71 if record.ttft is not None: 

72 update["completion_start_time"] = record.created + timedelta(milliseconds=record.ttft) 

73 if (usage := record.usage) is not None: 

74 update["usage_details"] = { 

75 # Langfuse sums every "input*" key into inputTokens, so "input" must exclude the cached tokens. 

76 "input": usage.prompt_tokens - usage.prompt_tokens_details.cached_tokens, 

77 "output": usage.completion_tokens, 

78 "input_cached_tokens": usage.prompt_tokens_details.cached_tokens, 

79 } 

80 update["cost_details"] = {"total": usage.cost} 

81 

82 return update
